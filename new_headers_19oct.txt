-- ============================================================================
-- 2026-10-15_PlantSpeciesIsActive.sql
--
-- WHY (owner requirement 2026-10-15, duplicate-variety cleanup): a duplicate variety must be RETIRED, never deleted --
--   its history moves to the kept variety and the row itself stays, marked inactive and renamed "DUP-...".
--   dbo.PlantSpecies has no active flag, so there is nothing to mark.
--
-- WHAT (one transaction; nothing else changes): adds dbo.PlantSpecies.IsActive BIT NOT NULL, default 1
--   (constraint DF_PlantSpecies_IsActive), so every existing and future variety starts active.
--
-- Safe to run more than once (no-op when the column already exists). Does not name a database:
-- select the target database first (SSMS: database dropdown / sqlcmd -d), then run.
-- Rollback: 2026-10-15_PlantSpeciesIsActive_Rollback.sql
-- ============================================================================
SET QUOTED_IDENTIFIER ON;
SET ANSI_NULLS ON;
SET NOCOUNT ON;
SET XACT_ABORT ON;

PRINT '=== Server / database: ' + @@SERVERNAME + ' / ' + DB_NAME() + ' ===';
IF OBJECT_ID(N'dbo.PlantSpecies', N'U') IS NULL
    THROW 53201, 'dbo.PlantSpecies not found -- wrong database selected?', 1;

IF COL_LENGTH(N'dbo.PlantSpecies', N'IsActive') IS NOT NULL
    PRINT '=== PlantSpecies.IsActive already exists -- no-op ===';
ELSE
BEGIN
    BEGIN TRANSACTION PlantSpeciesIsActive;
    BEGIN TRY
        ALTER TABLE dbo.PlantSpecies ADD IsActive BIT NOT NULL CONSTRAINT DF_PlantSpecies_IsActive DEFAULT (1);
        COMMIT TRANSACTION PlantSpeciesIsActive;
        PRINT '=== Added PlantSpecies.IsActive (all existing varieties = 1) ===';
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        THROW;
    END CATCH
END
GO
SELECT COUNT(*) AS Varieties, SUM(CASE WHEN IsActive = 1 THEN 1 ELSE 0 END) AS Active FROM dbo.PlantSpecies;

######## 2026-10-15_PlantSpeciesUniqueActiveName ########
-- ============================================================================
-- 2026-10-15_PlantSpeciesUniqueActiveName.sql
--
-- WHY (owner requirement 2026-10-15): after the duplicate-variety cleanup, the database itself must refuse a second
--   ACTIVE variety with the same name (the Add / Edit Variety page already refuses it; this also stops scripts / imports).
--
-- WHAT (one transaction; nothing else changes): unique index UX_PlantSpecies_Name_Active on dbo.PlantSpecies (Name)
--   WHERE IsActive = 1. The column's collation (SQL_Latin1_General_CP1_CI_AS) ignores capitals and trailing spaces;
--   retired "DUP-..." rows (IsActive = 0) are outside the index.
--
-- ORDER: run AFTER 2026-10-15_PlantSpeciesIsActive.sql AND Fix_Duplicate_Varieties.sql (@Commit = 1).
--   Refuses -- listing the clashing names -- while any two active varieties still share a name.
-- Safe to run more than once. Does not name a database: select the target database first.
-- Rollback: 2026-10-15_PlantSpeciesUniqueActiveName_Rollback.sql
-- ============================================================================
SET QUOTED_IDENTIFIER ON;
SET ANSI_NULLS ON;
SET ANSI_PADDING ON;
SET ANSI_WARNINGS ON;
SET ARITHABORT ON;
SET CONCAT_NULL_YIELDS_NULL ON;
SET NUMERIC_ROUNDABORT OFF;
SET NOCOUNT ON;
SET XACT_ABORT ON;

PRINT '=== Server / database: ' + @@SERVERNAME + ' / ' + DB_NAME() + ' ===';
IF COL_LENGTH(N'dbo.PlantSpecies', N'IsActive') IS NULL
    THROW 53221, 'dbo.PlantSpecies.IsActive is missing -- run 2026-10-15_PlantSpeciesIsActive.sql first.', 1;

IF EXISTS (SELECT 1 FROM sys.indexes WHERE object_id = OBJECT_ID(N'dbo.PlantSpecies') AND name = N'UX_PlantSpecies_Name_Active')
BEGIN
    PRINT '=== UX_PlantSpecies_Name_Active already exists -- no-op ===';
    RETURN;
END

DECLARE @Dups TABLE (Name NVARCHAR(100), Ids NVARCHAR(400));
EXEC sp_executesql N'SELECT MIN(Name), STRING_AGG(CAST(Id AS NVARCHAR(12)), N'', '') FROM dbo.PlantSpecies WHERE IsActive = 1 GROUP BY Name HAVING COUNT(*) > 1;';
INSERT @Dups EXEC sp_executesql N'SELECT MIN(Name), STRING_AGG(CAST(Id AS NVARCHAR(12)), N'', '') FROM dbo.PlantSpecies WHERE IsActive = 1 GROUP BY Name HAVING COUNT(*) > 1;';
IF EXISTS (SELECT 1 FROM @Dups)
    THROW 53222, 'Active varieties still share a name (listed above) -- run Fix_Duplicate_Varieties.sql first. Nothing created.', 1;

BEGIN TRANSACTION PlantSpeciesUniqueActiveName;
BEGIN TRY
    EXEC sp_executesql N'CREATE UNIQUE NONCLUSTERED INDEX UX_PlantSpecies_Name_Active ON dbo.PlantSpecies (Name) WHERE IsActive = 1;';
    COMMIT TRANSACTION PlantSpeciesUniqueActiveName;
    PRINT '=== Created UX_PlantSpecies_Name_Active (unique Name among active varieties) ===';
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
    THROW;
END CATCH

######## Fix_Duplicate_Varieties ########
-- ============================================================================
-- Fix_Duplicate_Varieties.sql   (2026-10-15)
--
-- Retires duplicate varieties in dbo.PlantSpecies and moves ALL their history to the kept variety.
-- No variety row and no transaction row is ever deleted.
--
--   @Commit = 0  PREVIEW (default): read-only. Shows the plan, the records that would move, the name / growing-day
--                changes and any BLOCKERS. Writes nothing.
--   @Commit = 1  SAVE: the same plan in ONE transaction; any error -> everything rolled back. Refused while any
--                blocker exists or while dbo.PlantSpecies.IsActive is missing (run 2026-10-15_PlantSpeciesIsActive.sql).
--
-- Does not name a database: select it first (SSMS database dropdown / sqlcmd -d). Matches by NAME, never by Id,
-- so the same file runs unchanged on local and live. Safe to run more than once (a second run finds nothing to do).
--
-- Owner decisions 2026-10-15:
--   * Each group below = the names that are one variety. KEEP = the OLDEST Id (lowest) among the active varieties whose
--     name matches any name of the group; it gets the group's final name. Every other match is a duplicate: its records
--     move to the kept Id, then it is set IsActive = 0 and renamed "DUP-<Id>-<old name>".
--     If only one match exists (e.g. live without this morning's rows), it is simply renamed.
--   * Names compare like Add Variety does: capitals ignored, spaces / tabs / line breaks at the ends ignored,
--     runs of spaces inside count as one space.
--   * All variety names are trimmed (spaces, tabs, line breaks at start / end).
--   * Every active "Lantana ..." variety gets Ready Stock Days = 25.
--   * "Marigold French Hot Pack Mix" is NOT part of this cleanup.
--
-- BLOCKERS (the script stops instead of guessing -- nothing is disabled, no trigger is bypassed):
--   - a group whose matches belong to different Plant Types
--   - a duplicate used in a table whose triggers forbid changing the variety (SeedSowings, PotProductionBatches,
--     OutletBookingItems) or by a child row of a composite (row + variety) foreign key
--   - a stock / balance row of a duplicate whose unique key (e.g. Area + Batch) already exists for the kept variety:
--     those quantities must be combined by hand first
--   - a bed sowing of a duplicate that its trigger would not let change variety (not untouched, or ledger mismatch)
--   - two different active varieties ending up with the same name after the cleanup
-- ============================================================================
SET QUOTED_IDENTIFIER ON;
SET ANSI_NULLS ON;
SET NOCOUNT ON;
SET XACT_ABORT ON;

DECLARE @Commit BIT = 0;          -- 0 = PREVIEW (read-only), 1 = SAVE

DECLARE @Ws NVARCHAR(10) = N' ' + NCHAR(9) + NCHAR(10) + NCHAR(13) + NCHAR(160);
DECLARE @HasIsActive BIT = CASE WHEN COL_LENGTH(N'dbo.PlantSpecies', N'IsActive') IS NULL THEN 0 ELSE 1 END;
DECLARE @sql NVARCHAR(MAX), @t SYSNAME, @c SYSNAME, @n INT;

IF OBJECT_ID(N'dbo.PlantSpecies', N'U') IS NULL
    THROW 53301, 'dbo.PlantSpecies not found -- wrong database selected?', 1;

SELECT @@SERVERNAME AS ServerName, DB_NAME() AS DatabaseName,
       CASE WHEN @Commit = 1 THEN 'SAVE (@Commit = 1)' ELSE 'PREVIEW (@Commit = 0) - read-only' END AS Mode,
       CASE WHEN @HasIsActive = 1 THEN 'present' ELSE 'MISSING - run 2026-10-15_PlantSpeciesIsActive.sql before saving' END AS PlantSpecies_IsActive;

DROP TABLE IF EXISTS #Spec, #Species, #Member, #Refs, #RefCount, #RefCountAfter, #TableTotal, #Blocker, #NameChange, #DaysChange;
